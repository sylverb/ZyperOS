# Developer mode on a release image (src/devmode.h): what is said when the
# unlock works, and the page MENU > Developer Mode opens. The row's own name is
# sysmenu.developer, Developer Mode, and its On or Off is menu.on_title and menu.off_title.

[devmode.is_on]
note = The line of the message that says the unlock worked. One line, which must fit the message panel: it does not wrap
en = Developer Mode is on
zh-Hans = 开发者模式已开启
es = Modo desarrollador activado
pt-BR = Modo desenvolvedor ativado
fr = Mode développeur activé
de = Entwicklermodus ist an
ja = 開発者モードがオンになりました
it = Modalità sviluppatore attiva

[devmode.where]
note = The second line of that message: what to do next. One line, which must fit the message panel: it does not wrap
en = Connect a computer.
zh-Hans = 请连接电脑。
es = Conecta una computadora.
pt-BR = Conecte um computador.
fr = Branchez un ordinateur.
de = Computer anschließen.
ja = パソコンをつないでください。
it = Collega un computer.

[devmode.wifi]
note = The MENU row under Developer Mode that lets SSH run while Wi-Fi is up, with keys only. On or Off, A switches it
en = SSH over Wi-Fi
zh-Hans = 通过 Wi-Fi 使用 SSH
es = SSH por Wi-Fi
pt-BR = SSH pelo Wi-Fi
fr = SSH en Wi-Fi
de = SSH über WLAN
ja = Wi-Fi で SSH
it = SSH via Wi-Fi

[devmode.wifi_said]
note = A message when SSH over Wi-Fi is turned on: only a key signs in over Wi-Fi, and how to get the first key on. Up to four balanced lines
en = Over Wi-Fi only a key signs in, never the password. Add your key over the cable, or in the card's Developer folder.
zh-Hans = 通过 Wi-Fi 只能用密钥登录，不能用密码。请先通过数据线添加你的密钥，或放进卡上的 Developer 文件夹。
es = Por Wi-Fi solo entra una clave, nunca la contraseña. Agrega tu clave por el cable o en la carpeta Developer de la tarjeta.
pt-BR = Pelo Wi-Fi só uma chave entra, nunca a senha. Adicione sua chave pelo cabo ou na pasta Developer do cartão.
fr = En Wi-Fi, seule une clé permet d'entrer, jamais le mot de passe. Ajoutez votre clé par le câble ou dans le dossier Developer.
de = Über WLAN meldet nur ein Schlüssel an, nie das Passwort. Füge deinen Schlüssel per Kabel oder im Ordner Developer hinzu.
ja = Wi-Fi では鍵でのみサインインでき、パスワードは使えません。ケーブル経由か、カードの Developer フォルダに鍵を追加してください。
it = Via Wi-Fi entra solo una chiave, mai la password. Aggiungi la tua chiave tramite il cavo o nella cartella Developer della scheda.

[devmode.password_is]
note = A line of the Developer page, over the cable: the password made for this plug, which signs in as root. %s is the password, eight letters and digits
en = Password: %s
zh-Hans = 密码：%s
es = Contraseña: %s
pt-BR = Senha: %s
fr = Mot de passe : %s
de = Passwort: %s
ja = パスワード：%s
it = Password: %s

en.one = %d key on the card
en.other = %d keys on the card
zh-Hans = 卡上有 %d 个密钥
es.one = %d clave en la tarjeta
es.other = %d claves en la tarjeta
pt-BR.one = %d chave no cartão
pt-BR.other = %d chaves no cartão
fr.one = %d clé sur la carte
fr.other = %d clés sur la carte
de.one = %d Schlüssel auf der Karte
de.other = %d Schlüssel auf der Karte
ja = カード上に鍵 %d 個
it.one = %d chiave sulla scheda
it.other = %d chiavi sulla scheda

[devmode.keys_in_folder_none]
